cmake_minimum_required(VERSION 3.20)
project(bloons LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bloons_core src/game.cpp src/levels.cpp)
target_include_directories(bloons_core PUBLIC src)
if(MSVC)
  target_compile_options(bloons_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(bloons_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  add_executable(bloons WIN32 src/windows.cpp)
  target_link_libraries(bloons PRIVATE bloons_core gdiplus winmm shell32)
  target_compile_definitions(bloons PRIVATE UNICODE _UNICODE NOMINMAX BLOONS_ASSET_ROOT=L"${CMAKE_CURRENT_SOURCE_DIR}")
  install(TARGETS bloons RUNTIME DESTINATION .)
  install(DIRECTORY frames sprites sounds fonts DESTINATION .)
  install(FILES LICENSE README.md PORTING.md DESTINATION .)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE bloons_core)
add_test(NAME core_tests COMMAND core_tests)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME swf_parser COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p test_swf.py)
  add_test(NAME rockbox_installer COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p test_rockbox.py)
endif()
if(WIN32)
  add_test(NAME render_and_input COMMAND bloons --render-check)
endif()

# Host regression checks for the firmware's bounded storage. Not an ARM build.
add_library(bloons_embedded_core src/game.cpp src/levels.cpp)
target_include_directories(bloons_embedded_core PUBLIC src)
target_compile_definitions(bloons_embedded_core PUBLIC BLOONS_EMBEDDED)
if(MSVC)
  target_compile_options(bloons_embedded_core PRIVATE /GR-)
else()
  target_compile_options(bloons_embedded_core PRIVATE -fno-exceptions -fno-rtti)
endif()
add_executable(embedded_core_tests tests/core_tests.cpp tests/embedded_failure.cpp)
target_link_libraries(embedded_core_tests PRIVATE bloons_embedded_core)
if(MSVC)
  target_link_options(embedded_core_tests PRIVATE /STACK:8388608)
endif()
add_test(NAME embedded_core_tests COMMAND embedded_core_tests)
add_executable(rockbox_adapter_tests tests/rockbox/adapter_tests.cpp ports/rockbox/bridge.cpp ports/rockbox/render.c)
target_include_directories(rockbox_adapter_tests PRIVATE ports/rockbox)
target_link_libraries(rockbox_adapter_tests PRIVATE bloons_embedded_core)
add_test(NAME rockbox_adapter_tests COMMAND rockbox_adapter_tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/rockbox/bloons.dat")
  add_test(NAME rockbox_assets COMMAND rockbox_adapter_tests "${CMAKE_CURRENT_SOURCE_DIR}/build/rockbox/bloons.dat")
endif()
